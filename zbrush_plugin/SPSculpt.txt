// ==============================================================================
// SPSculpt Companion Viewport - ZBrush Plugin (ZScript)
// ==============================================================================
// Installation:
//   Copy this file and SPSculptBridge.dll to:
//   C:\Program Files\Pixologic\ZBrush 202X\ZStartup\ZPlugs64\
//   or
//   C:\Program Files\Maxon ZBrush 202X\ZStartup\ZPlugs64\
//
// Description:
//   Enables lightning-fast synchronization between ZBrush and SPSculpt.
//   Supports both manual 'Sync Active' and automatic stroke end interception
//   via native in-process subclassing with SPSculptBridge.dll.
// ==============================================================================

[ISubPalette, "ZPlugin:SPSculpt"]

// Helper: DLL path resolver
[VarDef, spsculptDataDir, "C:/Users/Public/Documents/SPSculptData/"]
[VarDef, spsculptDllPath, ""]

// 1. Manual Sync Button
[IButton, "ZPlugin:SPSculpt:Sync Active", "Export active SubTool to SPSculpt Companion Viewport",
    [VarDef, meshFile, [StrMerge, [Var, spsculptDataDir], "mesh.GoZ"]]
    [VarDef, camFile, [StrMerge, [Var, spsculptDataDir], "camera.txt"]]

    // Perform native GoZ export
    [FileNameSetNext, [Var, meshFile]]
    [IPress, "Tool:Export"]

    // Export camera orientation snapshot
    [VarDef, xP, 0][VarDef, yP, 0][VarDef, zP, 0]
    [VarDef, xR, 0][VarDef, yR, 0][VarDef, zR, 0]
    [VarDef, sc, 0]
    [TransformGet, xP, yP, zP, xR, yR, zR, sc]

    [MemCreate, SPSculptCamMem, 256, 0]
    [MemWriteString, SPSculptCamMem, [StrMerge, [Var, xP], ",", [Var, yP], ",", [Var, zP], ",", [Var, xR], ",", [Var, yR], ",", [Var, zR], ",", [Var, sc], "\n"], 0]
    [MemSaveToFile, SPSculptCamMem, [Var, camFile], 1]
    [MemDelete, SPSculptCamMem]

    [Note, [StrMerge, "SPSculpt: Synced active tool to ", [Var, meshFile]], , 1]
, , 0.5]

// 2. Enable Auto-Sync Hook (In-Process DLL Subclassing)
[IButton, "ZPlugin:SPSculpt:Enable Auto-Sync", "Install in-process hook for automatic stroke-end export",
    [VarSet, spsculptDllPath, [FileNameResolvePath, "SPSculptBridge.dll"]]
    [VarDef, res, 0]
    [VarSet, res, [FileExecute, [Var, spsculptDllPath], "InstallHook"]]
    [If, [Var, res] >= 1,
        [FileExecute, [Var, spsculptDllPath], "SetDataDir", [Var, spsculptDataDir]]
        [Note, "SPSculpt: Auto-Sync Hook INSTALLED! Changes will sync on brush stroke completion.", , 2]
    ,
        [Note, [StrMerge, "SPSculpt: Failed to install hook. Make sure SPSculptBridge.dll is in ZPlugs64. Path: ", [Var, spsculptDllPath]], , 3]
    ]
, , 0.5]

// 3. Disable Auto-Sync Hook
[IButton, "ZPlugin:SPSculpt:Disable Auto-Sync", "Remove in-process hook",
    [VarSet, spsculptDllPath, [FileNameResolvePath, "SPSculptBridge.dll"]]
    [VarDef, res2, 0]
    [VarSet, res2, [FileExecute, [Var, spsculptDllPath], "RemoveHook"]]
    [Note, "SPSculpt: Auto-Sync Hook REMOVED.", , 1]
, , 0.5]
