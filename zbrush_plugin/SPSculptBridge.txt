[ISubPalette, "ZPlugin:SPSculpt"]

[VarDef, spsculptDataDir, "C:/Users/Public/Documents/SPSculptData/"]
[VarDef, spsculptDllPath, ""]
[VarDef, autoSyncActive, 0]
[VarDef, meshFile, ""]
[VarDef, lastUndoCounter, -999999]
[VarDef, curUndoCounter, 0]
[VarDef, userThrottleMs, 800]
[VarDef, canExport, 1]
[VarDef, skipCount, 0]

// Helper routine: resolves SPSculptBridge.dll with absolute, exchange and local probes
[RoutineDef, FindBridgeDll,
    [VarSet, spsculptDllPath, ""]
    [VarDef, probe, ""]
    // 1. Direct workspace location
    [If, [FileExists, "C:/Users/user/Desktop/cpp/sculptsp-native/zbrush_plugin/SPSculptBridgeData/SPSculptBridge.dll"],
        [VarSet, spsculptDllPath, "C:/Users/user/Desktop/cpp/sculptsp-native/zbrush_plugin/SPSculptBridgeData/SPSculptBridge.dll"]
    ,
        // 2. Default public exchange directory
        [If, [FileExists, "C:/Users/Public/Documents/SPSculptData/SPSculptBridge.dll"],
            [VarSet, spsculptDllPath, "C:/Users/Public/Documents/SPSculptData/SPSculptBridge.dll"]
        ,
            // 3. Official ZBrush convention: SPSculptBridgeData/SPSculptBridge.dll
            [VarSet, probe, [FileNameResolvePath, "SPSculptBridgeData/SPSculptBridge.dll"]]
            [If, [FileExists, [Var, probe]],
                [VarSet, spsculptDllPath, [Var, probe]]
            ,
                // Fallback
                [VarSet, spsculptDllPath, [FileNameResolvePath, "SPSculptBridge.dll"]]
            ]
        ]
    ]
]

// ------------------------------------------------------------------------------
// Inbound Import Routines (SPSculpt -> ZBrush)
// ------------------------------------------------------------------------------

[RoutineDef, ImportActiveMesh,
    [If, [IsEnabled, "Tool:Import"],
        [VarDef, activeGozPath, ""]
        [VarSet, activeGozPath, [StrMerge, [Var, spsculptDataDir], "out_subtools/mesh_active.GoZ"]]
        [If, [FileExists, [Var, activeGozPath]],
            [FileNameSetNext, [Var, activeGozPath]]
            [IPress, "Tool:Import"]
            [VarDef, pActiveCmd, ""]
            [VarSet, pActiveCmd, [StrMerge, [Var, spsculptDataDir], "cmd_import_active.txt"]]
            [If, [FileExists, [Var, pActiveCmd]],
                [FileDelete, [Var, pActiveCmd]]
            ]
            [Note, "SPSculpt: Active mesh imported into current SubTool!", , 1]
        ,
            [Note, [StrMerge, "SPSculpt: No mesh found at ", [Var, activeGozPath]], , 2]
        ]
    ,
        [Note, "SPSculpt: Tool:Import is disabled. Please enter 3D Edit mode.", , 2]
    ]
]

[RoutineDef, ImportSceneRoutine,
    [VarDef, pSceneCmd, ""]
    [VarSet, pSceneCmd, [StrMerge, [Var, spsculptDataDir], "cmd_import_scene.txt"]]
    [If, [FileExists, [Var, pSceneCmd]],
        [FileNameSetNext, [Var, pSceneCmd]]
        [IPress, "ZScript:Load"]
        [FileDelete, [Var, pSceneCmd]]
    ,
        [Note, [StrMerge, "SPSculpt: No scene script found at ", [Var, pSceneCmd]], , 2]
    ]
]

// Background check for pending imports from SPSculpt
[RoutineDef, CheckInboundImports,
    [VarDef, chkAct, ""]
    [VarSet, chkAct, [StrMerge, [Var, spsculptDataDir], "cmd_import_active.txt"]]
    [If, [FileExists, [Var, chkAct]],
        [RoutineCall, ImportActiveMesh]
    ]

    [VarDef, chkScn, ""]
    [VarSet, chkScn, [StrMerge, [Var, spsculptDataDir], "cmd_import_scene.txt"]]
    [If, [FileExists, [Var, chkScn]],
        [RoutineCall, ImportSceneRoutine]
    ]
]

// Helper routine: checks Undo Counter and exports mesh only if geometry changed + throttled
[RoutineDef, CheckAndExportIfChanged,
    [If, [Var, autoSyncActive] == 1,
        // 1. Query current undo history step
        [VarSet, curUndoCounter, 0]
        [If, [IsEnabled, "Edit:Tool:Undo Counter"],
            [VarSet, curUndoCounter, [IGet, "Edit:Tool:Undo Counter"]]
        ]

        // 2. Export ONLY if Undo Counter changed (actual brush stroke, smooth, or undo/redo)
        [If, [Var, curUndoCounter] != [Var, lastUndoCounter],
            // 3. Time throttle check via DLL (e.g. min 800ms between disk writes)
            [VarSet, canExport, 1]
            [If, [FileExists, [Var, spsculptDllPath]],
                [VarSet, canExport, [FileExecute, [Var, spsculptDllPath], "CanExportNow", "", [Var, userThrottleMs]]]
            ]

            [VarSet, skipCount, [Var, skipCount] + 1]
            // Allow export if throttle time passed OR if 4 quick strokes were skipped
            [If, ([Var, canExport] == 1) || ([Var, skipCount] >= 4),
                [VarSet, skipCount, 0]
                [VarSet, lastUndoCounter, [Var, curUndoCounter]]
                [If, [IsEnabled, "Tool:Export"],
                    [VarSet, meshFile, [StrMerge, [Var, spsculptDataDir], "mesh.GoZ"]]
                    [FileNameSetNext, [Var, meshFile]]
                    [IPress, "Tool:Export"]
                ]
            ]
        ]
    ]
]

// Full Scene Export Routine (ZBrush -> SPSculpt)
[RoutineDef, ExportFullScene,
    [If, [IsEnabled, "Tool:Export"],
        [RoutineCall, FindBridgeDll]
        [VarDef, totalSubTools, 0]
        [VarSet, totalSubTools, [SubToolGetCount]]
        [VarDef, origActiveIdx, 0]
        [VarSet, origActiveIdx, [SubToolGetActiveIndex]]

        [If, [FileExists, [Var, spsculptDllPath]],
            [FileExecute, [Var, spsculptDllPath], "BeginSceneManifest", [Var, spsculptDataDir]]
        ]

        [VarDef, stStatus, 0]
        [VarDef, isVisible, 0]
        [VarDef, shouldExport, 1]
        [VarDef, rawTitle, ""]
        [VarDef, subName, ""]
        [VarDef, titleLen, 0]
        [VarDef, curSubUndo, 0]
        [VarDef, isDirty, 1]
        [VarDef, subGozRelPath, ""]
        [VarDef, subGozFullPath, ""]
        [VarDef, manifestEntry, ""]

        [IFreeze,
            [Loop, totalSubTools,
                [VarSet, stStatus, [SubToolGetStatus, n]]
                [If, (stStatus & 1) == 1,
                    [VarSet, isVisible, 1]
                ,
                    [VarSet, isVisible, 0]
                ]

                [VarSet, shouldExport, 1]
                [If, (exportOnlyVisible == 1) && (isVisible == 0),
                    [VarSet, shouldExport, 0]
                ]

                [If, shouldExport == 1,
                    [SubToolSelect, n]
                    [VarSet, rawTitle, [IGetTitle, "Tool:ItemInfo"]]
                    [VarSet, titleLen, [StrLength, rawTitle]]
                    [If, titleLen > 1,
                        [VarSet, subName, [StrExtract, rawTitle, 0, titleLen - 2]]
                    ,
                        [VarSet, subName, [StrMerge, "SubTool_", n]]
                    ]

                    [VarSet, curSubUndo, 0]
                    [If, [IsEnabled, "Edit:Tool:Undo Counter"],
                        [VarSet, curSubUndo, [IGet, "Edit:Tool:Undo Counter"]]
                    ]

                    // Dirty check via DLL
                    [VarSet, isDirty, 1]
                    [If, [FileExists, [Var, spsculptDllPath]],
                        [VarSet, isDirty, [FileExecute, [Var, spsculptDllPath], "CheckSubToolDirty", subName, curSubUndo]]
                    ]

                    [VarSet, subGozRelPath, [StrMerge, "subtools/subtool_", n, ".GoZ"]]
                    [VarSet, subGozFullPath, [StrMerge, [Var, spsculptDataDir], subGozRelPath]]

                    [If, isDirty == 1,
                        [FileNameSetNext, subGozFullPath]
                        [IPress, "Tool:Export"]
                    ]

                    [If, [FileExists, [Var, spsculptDllPath]],
                        // Format: "id|visible|undoCounter|relPath|subName"
                        [VarSet, manifestEntry, [StrMerge, n, "|", isVisible, "|", curSubUndo, "|", subGozRelPath, "|", subName]]
                        [FileExecute, [Var, spsculptDllPath], "AddManifestEntry", manifestEntry]
                    ]
                ]
            , n]

            // Restore original active subtool
            [SubToolSelect, origActiveIdx]
        ]

        [If, [FileExists, [Var, spsculptDllPath]],
            [FileExecute, [Var, spsculptDllPath], "SaveSceneManifest", "", origActiveIdx]
        ]
        [Note, [StrMerge, "SPSculpt: Scene synced! (", totalSubTools, " SubTools)"], , 1]
    ,
        [Note, "SPSculpt: Tool:Export is disabled. Please enter 3D Edit mode.", , 2]
    ]
    ,
    exportOnlyVisible
]

// Background routine: triggered on timer or stroke release (Timer = 1, Mouse Up = 8, Mouse Move = 2)
[RoutineDef, AutoSyncHandler,
    [If, [Var, autoSyncActive] == 1,
        [RoutineCall, CheckInboundImports]
        [RoutineCall, CheckAndExportIfChanged]
        // Continue listening: wake up every 1 second or on user interaction
        [SleepAgain, 1, 11]
    ]
]

// Internal button callable by DLL
[IButton, "ZPlugin:SPSculpt:CheckSync", "Trigger conditional sync if undo step changed",
    [RoutineCall, CheckInboundImports]
    [RoutineCall, CheckAndExportIfChanged]
, , 0.01]

[ISlider, "ZPlugin:SPSculpt:Sync Throttle", 800, 50, 200, 2000, "Minimum delay between mesh exports in ms (prevents stutter)",
    [VarSet, userThrottleMs, [IGet, "ZPlugin:SPSculpt:Sync Throttle"]]
, , 0.5]

[IButton, "ZPlugin:SPSculpt:Enable Auto-Sync", "Enable automatic bi-directional auto-sync",
    [If, [IsEnabled, "ZPlugin:SPSculpt:Sync Throttle"],
        [VarSet, userThrottleMs, [IGet, "ZPlugin:SPSculpt:Sync Throttle"]]
    ]
    [RoutineCall, FindBridgeDll]
    [If, [FileExists, [Var, spsculptDllPath]],
        [FileExecute, [Var, spsculptDllPath], "InstallHook"]
        [FileExecute, [Var, spsculptDllPath], "SetDataDir", [Var, spsculptDataDir]]
    ]
    [If, [IsEnabled, "Edit:Tool:Undo Counter"],
        [VarSet, lastUndoCounter, [IGet, "Edit:Tool:Undo Counter"]]
    ,
        [VarSet, lastUndoCounter, 0]
    ]
    [VarSet, skipCount, 0]
    [VarSet, autoSyncActive, 1]
    [Note, [StrMerge, "SPSculpt: Bi-Directional Auto-Sync ENABLED! (Throttle: ", [Var, userThrottleMs], " ms)"], , 2]
    // Check immediately for any pending inbound imports
    [RoutineCall, CheckInboundImports]
    // Launch active event listener on timer + interaction
    [Sleep, 1, [RoutineCall, AutoSyncHandler], 11]
, , 0.5]

[IButton, "ZPlugin:SPSculpt:Disable Auto-Sync", "Disable automatic stroke-end export",
    [VarSet, autoSyncActive, 0]
    [VarSet, lastUndoCounter, -999999]
    [RoutineCall, FindBridgeDll]
    [If, [FileExists, [Var, spsculptDllPath]],
        [FileExecute, [Var, spsculptDllPath], "RemoveHook"]
    ]
    [Note, "SPSculpt: Auto-Sync DISABLED.", , 1]
, , 0.5]
