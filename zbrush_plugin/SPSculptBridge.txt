[ISubPalette, "ZPlugin:SPSculpt"]

[VarDef, spsculptDataDir, "C:/Users/Public/Documents/SPSculptData/"]
[VarDef, spsculptDllPath, ""]
[VarDef, autoSyncActive, 0]
[VarDef, meshFile, ""]
[VarDef, lastUndoCounter, -999999]
[VarDef, curUndoCounter, 0]
[VarDef, userThrottleMs, 800]
[VarDef, canExport, 1]
[VarDef, skipCount, 0]

// Helper routine: resolves SPSculptBridge.dll with absolute, exchange and local probes
[RoutineDef, FindBridgeDll,
    [VarSet, spsculptDllPath, ""]
    [VarDef, probe, ""]
    // 1. Direct workspace location
    [If, [FileExists, "C:/Users/user/Desktop/cpp/sculptsp-native/zbrush_plugin/SPSculptBridgeData/SPSculptBridge.dll"],
        [VarSet, spsculptDllPath, "C:/Users/user/Desktop/cpp/sculptsp-native/zbrush_plugin/SPSculptBridgeData/SPSculptBridge.dll"]
    ,
        // 2. Default public exchange directory
        [If, [FileExists, "C:/Users/Public/Documents/SPSculptData/SPSculptBridge.dll"],
            [VarSet, spsculptDllPath, "C:/Users/Public/Documents/SPSculptData/SPSculptBridge.dll"]
        ,
            // 3. Official ZBrush convention: SPSculptBridgeData/SPSculptBridge.dll
            [VarSet, probe, [FileNameResolvePath, "SPSculptBridgeData/SPSculptBridge.dll"]]
            [If, [FileExists, [Var, probe]],
                [VarSet, spsculptDllPath, [Var, probe]]
            ,
                // Fallback
                [VarSet, spsculptDllPath, [FileNameResolvePath, "SPSculptBridge.dll"]]
            ]
        ]
    ]
]

// Helper routine: checks Undo Counter and exports mesh only if geometry changed + throttled
[RoutineDef, CheckAndExportIfChanged,
    [If, [Var, autoSyncActive] == 1,
        // 1. Query current undo history step
        [VarSet, curUndoCounter, 0]
        [If, [IsEnabled, "Edit:Tool:Undo Counter"],
            [VarSet, curUndoCounter, [IGet, "Edit:Tool:Undo Counter"]]
        ]

        // 2. Export ONLY if Undo Counter changed (actual brush stroke, smooth, or undo/redo)
        // Camera navigation (orbit/pan/zoom) and key presses do NOT change Undo Counter!
        [If, [Var, curUndoCounter] != [Var, lastUndoCounter],
            // 3. Time throttle check via DLL (e.g. min 800ms between disk writes)
            [VarSet, canExport, 1]
            [If, [FileExists, [Var, spsculptDllPath]],
                [VarSet, canExport, [FileExecute, [Var, spsculptDllPath], "CanExportNow", , [Var, userThrottleMs]]]
            ]

            [VarSet, skipCount, [Var, skipCount] + 1]
            // Allow export if throttle time passed OR if 4 quick strokes were skipped
            [If, ([Var, canExport] == 1) || ([Var, skipCount] >= 4),
                [VarSet, skipCount, 0]
                [VarSet, lastUndoCounter, [Var, curUndoCounter]]
                [If, [IsEnabled, "Tool:Export"],
                    [VarSet, meshFile, [StrMerge, [Var, spsculptDataDir], "mesh.GoZ"]]
                    [FileNameSetNext, [Var, meshFile]]
                    [IPress, "Tool:Export"]
                ]
            ]
        ]
    ]
]

// Background routine: triggered reliably on each stroke release (LButton Up = event 8)
[RoutineDef, AutoSyncHandler,
    [If, [Var, autoSyncActive] == 1,
        [RoutineCall, CheckAndExportIfChanged]
        // Continue listening for next brush stroke release (LButton Up = 8)
        [SleepAgain, 3600, 8]
    ]
]

// Internal button callable by DLL via WM_COPYDATA
[IButton, "ZPlugin:SPSculpt:CheckSync", "Trigger conditional sync if undo step changed",
    [RoutineCall, CheckAndExportIfChanged]
, , 0.01]

// 1. Manual Sync Button
[IButton, "ZPlugin:SPSculpt:Sync Active", "Export active SubTool to SPSculpt Companion Viewport",
    [If, [IsEnabled, "Tool:Export"],
        [VarSet, meshFile, [StrMerge, [Var, spsculptDataDir], "mesh.GoZ"]]
        [FileNameSetNext, [Var, meshFile]]
        [IPress, "Tool:Export"]
        [If, [IsEnabled, "Edit:Tool:Undo Counter"],
            [VarSet, lastUndoCounter, [IGet, "Edit:Tool:Undo Counter"]]
        ]
        [If, [FileExists, [Var, spsculptDllPath]],
            [FileExecute, [Var, spsculptDllPath], "MarkExportDone"]
        ]
        [Note, [StrMerge, "SPSculpt: Synced active tool to ", [Var, meshFile]], , 1]
    ,
        [Note, "SPSculpt: Tool:Export is disabled. Please enter 3D Edit mode.", , 2]
    ]
, , 0.5]

// 2. Throttle Interval Slider (200ms - 2000ms)
[ISlider, "ZPlugin:SPSculpt:Sync Throttle", 800, 50, 200, 2000, "Minimum delay between mesh exports in ms (prevents stutter)",
    [VarSet, userThrottleMs, [IGet, "ZPlugin:SPSculpt:Sync Throttle"]]
, , 0.5]

// 3. Enable Auto-Sync (100% reliable stroke-end export via native Sleep event 8)
[IButton, "ZPlugin:SPSculpt:Enable Auto-Sync", "Enable automatic stroke-end export with throttling",
    [If, [IsEnabled, "ZPlugin:SPSculpt:Sync Throttle"],
        [VarSet, userThrottleMs, [IGet, "ZPlugin:SPSculpt:Sync Throttle"]]
    ]
    [RoutineCall, FindBridgeDll]
    [If, [FileExists, [Var, spsculptDllPath]],
        [FileExecute, [Var, spsculptDllPath], "InstallHook"]
        [FileExecute, [Var, spsculptDllPath], "SetDataDir", [Var, spsculptDataDir]]
    ]
    // Baseline current undo counter so enabling auto-sync doesn't trigger immediate fake export
    [If, [IsEnabled, "Edit:Tool:Undo Counter"],
        [VarSet, lastUndoCounter, [IGet, "Edit:Tool:Undo Counter"]]
    ,
        [VarSet, lastUndoCounter, 0]
    ]
    [VarSet, skipCount, 0]
    [VarSet, autoSyncActive, 1]
    [Note, [StrMerge, "SPSculpt: Auto-Sync ENABLED! (Throttle: ", [Var, userThrottleMs], " ms)"], , 2]
    // Launch active event listener on stroke release (LButton Up = 8) - 100% reliable in ZBrush!
    [Sleep, 3600, [RoutineCall, AutoSyncHandler], 8]
, , 0.5]

// 4. Disable Auto-Sync
[IButton, "ZPlugin:SPSculpt:Disable Auto-Sync", "Disable automatic stroke-end export",
    [VarSet, autoSyncActive, 0]
    [VarSet, lastUndoCounter, -999999]
    [RoutineCall, FindBridgeDll]
    [If, [FileExists, [Var, spsculptDllPath]],
        [FileExecute, [Var, spsculptDllPath], "RemoveHook"]
    ]
    [Note, "SPSculpt: Auto-Sync DISABLED.", , 1]
, , 0.5]
